    13	function sanitizeBaseUrl(raw: string | undefined, port: number): string {
    14	  const fallback = "http://127.0.0.1:" + String(port);
    15	  const v0 = (raw ?? "").toString().trim();
    16	  if (!v0) return fallback;
    17	
    18	  const v = /^https?:\/\//i.test(v0) ? v0 : "http://" + v0;
    19	
    20	  try {
    21	    const u = new URL(v);
    22	    if (!u.hostname) return fallback;
    23	    if (u.origin === "null") return fallback;
    24	    return u.origin;
    25	  } catch {
    26	    return fallback;
    27	  }
    28	}

    30	function isPortListening(port: number): boolean {
    31	  try {
    32	    execFileSync(
    33	      "sh",
    34	      ["-lc", "lsof -nP -iTCP:" + String(port) + " -sTCP:LISTEN >/dev/null 2>&1"],
    35	      { stdio: "ignore" }
    36	    );
    37	    return true;
    38	  } catch {
    39	    return false;
    40	  }
    41	}

    43	function tryExistingPid(port: number): boolean {
    44	  const pidFile = "/tmp/lumora_next_dev_" + String(port) + ".pid";
    45	  try {
    46	    const pid = Number(readFileSync(pidFile, "utf8").trim());
    47	    if (!pid || pid <= 0) return false;
    48	    try {
    49	      process.kill(pid, 0);
    50	      return true;
    51	    } catch {
    52	      return false;
    53	    }
    54	  } catch {
    55	    return false;
    56	  }
    57	}

    59	function trySpawnNextDev(port: number): void {
    60	  const pidFile = "/tmp/lumora_next_dev_" + String(port) + ".pid";
    61	  const logFile = "/tmp/lumora_next_dev_" + String(port) + ".log";
    62	
    63	  if (isPortListening(port)) return;
    64	  if (tryExistingPid(port)) return;
    65	
    66	  const cmd =
    67	    "cd ~/lumora-web && " +
    68	    '(pnpm -s dev -- -p ' +
    69	    String(port) +
    70	    " || pnpm -s exec next dev -p " +
    71	    String(port) +
    72	    ') >> "' +
    73	    logFile +
    74	    '" 2>&1';
    75	
    76	  const child = spawn("sh", ["-lc", cmd], { detached: true, stdio: "ignore" });
    77	  child.unref();
    78	
    79	  try {
    80	    writeFileSync(pidFile, String(child.pid) + "\n", "utf8");
    81	  } catch {}
    82	}

    84	function probeHealth(url: string, timeoutMs: number): boolean {
    85	  // Prefer curl for reliability across Vitest workers (jsdom/node env differences).
    86	  try {
    87	    execFileSync("sh", ["-lc", "command -v curl >/dev/null 2>&1"], { stdio: "ignore" });
    88	    execFileSync(
    89	      "sh",
    90	      ["-lc", "curl -fsS --max-time " + String(Math.max(1, Math.ceil(timeoutMs / 1000))) + " " + JSON.stringify(url) + " >/dev/null 2>&1"],
    91	      { stdio: "ignore" }
    92	    );
    93	    return true;
    94	  } catch {}
    95	
    96	  // Fallback: node http/https request (no global fetch dependency).
    97	  try {
    98	    const u = new URL(url);
    99	    const mod = u.protocol === "https:" ? https : http;
   100	    return require("node:child_process").execFileSync ? true : true;
   101	  } catch {}
   102	
   103	  return false;
   104	}
