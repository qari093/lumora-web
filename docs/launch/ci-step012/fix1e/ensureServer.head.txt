import { spawn } from "node:child_process";

const DEFAULT_BASE = "http://127.0.0.1:3000";
const DEFAULT_PORT = 3000;

function sleep(ms: number) {
  return new Promise((r) => setTimeout(r, ms));
}

function now() {
  return Date.now();
}

function baseUrl(): string {
  const raw = (process.env.BASE_URL || "").trim();
  if (!raw) return DEFAULT_BASE;
  return raw;
}

async function fetchOk(path: string, timeoutMs: number): Promise<boolean> {
  const BASE = baseUrl();
  const ac = new AbortController();
  const t = setTimeout(() => ac.abort(new Error(`abort:${timeoutMs}ms`)), timeoutMs);
  try {
    const res = await fetch(new URL(path, BASE), {
      cache: "no-store",
      redirect: "manual",
      signal: ac.signal,
    });
    if (!res.ok) return false;
    const ct = res.headers.get("content-type") ?? "";
    if (!ct.includes("application/json")) return false;
    const text = await res.text();
    try {
      const j = JSON.parse(text);
      return !!j?.ok;
    } catch {
      return false;
    }
  } catch {
    return false;
  } finally {
    clearTimeout(t);
  }
}

function startNextDevBestEffort(): void {
  // never await; never hang. best-effort spawn detached.
  const port = Number(process.env.PORT || DEFAULT_PORT);
  const log = `/tmp/lumora_next_dev_${port}.log`;

  // Prefer pnpm if present, fall back to npx.
  const cmd =
    process.platform === "win32"
      ? "cmd"
      : "sh";

  const args =
    process.platform === "win32"
      ? ["/c", `set PORT=${port} && pnpm -s dev > ${log} 2>&1`]
      : ["-lc", `PORT=${port} (command -v pnpm >/dev/null 2>&1 && pnpm -s dev || npx --yes next dev -p ${port}) >>"${log}" 2>&1 & disown || true`];

  try {
    spawn(cmd, args, { stdio: "ignore", detached: true }).unref();
  } catch {
    // ignore
  }
}

export async function ensureServerReady(opts?: { maxWaitMs?: number }) {
  const maxWaitMs = Math.max(1_000, Math.min(opts?.maxWaitMs ?? 20_000, 25_000));
  const deadline = now() + maxWaitMs;

  // Fast path
  if (await fetchOk("/api/health", 1_800)) return;

  // Try starting server once (best-effort) then poll
  startNextDevBestEffort();

  while (now() < deadline) {
    if (await fetchOk("/api/health", 2_500)) return;
    await sleep(350);
  }

  throw new Error(`Test server not healthy within timeout (${maxWaitMs}ms) @ ${baseUrl()}`);
}
