import { spawn } from "node:child_process";

const DEFAULT_BASE = "http://127.0.0.1:3000";
const DEFAULT_PORT = 3000;

function sleep(ms: number) {
  return new Promise((r) => setTimeout(r, ms));
}

function now() {
  return Date.now();
}

function sanitizeBase(raw: string): string | null {
  const v = (raw || "").trim();
  if (!v) return null;
  // Reject relative bases like "/" or "127.0.0.1:3000" (missing scheme)
  if (!(v.startsWith("http://") || v.startsWith("https://"))) return null;
  try {
    const u = new URL(v);
    if (!u.hostname) return null;
    return u.toString().replace(/\/+$/, ""); // no trailing slash
  } catch {
    return null;
  }
}

function baseUrlResolved(): string {
  const env = sanitizeBase(process.env.BASE_URL || "");
  return env ?? DEFAULT_BASE;
}

async function fetchOk(path: string, timeoutMs: number): Promise<boolean> {
  const BASE = baseUrlResolved();
  const ac = new AbortController();
  const t = setTimeout(() => ac.abort(new Error(`abort:${timeoutMs}ms`)), timeoutMs);
  try {
    const res = await fetch(new URL(path, BASE), {
      cache: "no-store",
      redirect: "manual",
      signal: ac.signal,
    });
    if (!res.ok) return false;
    const ct = res.headers.get("content-type") ?? "";
    if (!ct.includes("application/json")) return false;
    const text = await res.text();
    try {
      const j = JSON.parse(text);
      return !!j?.ok;
    } catch {
      return false;
    }
  } catch {
    return false;
  } finally {
    clearTimeout(t);
  }
}

function startNextDevBestEffort(): void {
  // Never await; best-effort spawn detached with minimal shell dependence.
  const port = Number(process.env.PORT || DEFAULT_PORT);
  const log = `/tmp/lumora_next_dev_${port}.log`;

  // Prefer pnpm if present, else npx next dev.
  const shell = process.platform === "win32" ? "cmd" : "sh";
  const cmd =
    process.platform === "win32"
      ? ["/c", `set PORT=${port} && (pnpm -s dev || npx --yes next dev -p ${port}) >> ${log} 2>&1`]
      : ["-lc", `PORT=${port} (command -v pnpm >/dev/null 2>&1 && pnpm -s dev || npx --yes next dev -p ${port}) >>"${log}" 2>&1 &`];

  try {
    spawn(shell, cmd, { stdio: "ignore", detached: true }).unref();
  } catch {
    // ignore
  }
}

export async function ensureServerReady(opts?: { maxWaitMs?: number }) {
  const maxWaitMs = Math.max(1_000, Math.min(opts?.maxWaitMs ?? 20_000, 25_000));
  const deadline = now() + maxWaitMs;

  // Fast path
  if (await fetchOk("/api/health", 2_000)) return;

  // Try starting server once (best-effort) then poll
  startNextDevBestEffort();

  while (now() < deadline) {
    if (await fetchOk("/api/health", 2_800)) return;
    await sleep(350);
  }

  throw new Error(`Test server not healthy within timeout (${maxWaitMs}ms) @ ${baseUrlResolved()}`);
}
