    11	function sanitizeBaseUrl(raw: string | undefined, port: number): string {
    12	  const v = (raw || "").trim();
    13	  if (!v) return \`http://127.0.0.1:\${port}\`;
    14	  if (!/^https?:\\/\\//i.test(v)) return \`http://\${v}\`;
    15	  return v;
    16	}

    18	function isPortListening(port: number): boolean {
    19	  try {
    20	    execFileSync("sh", ["-lc", \`lsof -nP -iTCP:\${port} -sTCP:LISTEN >/dev/null 2>&1\`], { stdio: "ignore" });
    21	    return true;
    22	  } catch {
    23	    return false;
    24	  }
    25	}

    27	function tryExistingPid(port: number): boolean {
    28	  const pidFile = \`/tmp/lumora_next_dev_\${port}.pid\`;
    29	  try {
    30	    const pid = Number(readFileSync(pidFile, "utf8").trim());
    31	    if (!pid || pid <= 0) return false;
    32	    try {
    33	      process.kill(pid, 0);
    34	      return true;
    35	    } catch {
    36	      return false;
    37	    }
    38	  } catch {
    39	    return false;
    40	  }
    41	}

    43	function trySpawnNextDev(port: number): void {
    44	  const pidFile = \`/tmp/lumora_next_dev_\${port}.pid\`;
    45	  const logFile = \`/tmp/lumora_next_dev_\${port}.log\`;
    46	
    47	  if (isPortListening(port)) return;
    48	  if (tryExistingPid(port)) return;
    49	
    50	  // Build a single shell command string (no nested template literals in generator).
    51	  const shell = [
    52	    "cd ~/lumora-web",
    53	    // Prefer pnpm dev; fallback to pnpm exec next dev.
    54	    \`(pnpm -s dev -- -p \${port} || pnpm -s exec next dev -p \${port}) >> "\${logFile}" 2>&1\`,
    55	  ].join(" && ");
    56	
    57	  const child = spawn("sh", ["-lc", shell], { detached: true, stdio: "ignore" });
    58	  child.unref();
    59	
    60	  try {
    61	    writeFileSync(pidFile, String(child.pid) + "\\n", "utf8");
    62	  } catch {}
    63	}
