
// Exported helper used by unit tests
export type LumaSpaceMode = "demo" | "beta" | "live" | (string & {});
export function deriveModeLabel(mode: LumaSpaceMode): string {
  const m = (mode ?? "").toString().toLowerCase();
  if (m === "demo") return "Demo";
  if (m === "beta") return "Beta";
  if (m === "live") return "Live";
  return m ? m.charAt(0).toUpperCase() + m.slice(1) : "";
}

export function LumaSpaceStateBanner() {
  return null;
}
export default LumaSpaceStateBanner;
