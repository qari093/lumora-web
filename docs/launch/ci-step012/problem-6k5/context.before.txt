    43	function tryExistingPid(port: number): boolean {
    44	  const pidFile = "/tmp/lumora_next_dev_" + String(port) + ".pid";
    45	  try {
    46	    const pid = Number(readFileSync(pidFile, "utf8").trim());
    47	    if (!pid || pid <= 0) return false;
    48	    try {
    49	      process.kill(pid, 0);
    50	      return true;
    51	    } catch {
    52	      return false;
    53	    }
    54	  } catch {
    55	    return false;
    56	  }
    57	}

    59	function trySpawnNextDev(port: number): void {
    60	  const pidFile = "/tmp/lumora_next_dev_" + String(port) + ".pid";
    61	  const logFile = "/tmp/lumora_next_dev_" + String(port) + ".log";
    62	
    63	  if (isPortListening(port)) return;
    64	  if (tryExistingPid(port)) return;
    65	
    66	  const cmd =
    67	    "cd ~/lumora-web && " +
    68	    '(PORT=' + String(port) +
    69	    ' pnpm -s dev || PORT=' + String(port) +
    70	    " pnpm -s exec next dev .) >> \"' +
    71	    logFile +
    72	    '\" 2>&1';
    73	
    74	  const child = spawn("sh", ["-lc", cmd], { detached: true, stdio: "ignore" });
    75	  child.unref();
    76	
    77	  try {
    78	    writeFileSync(pidFile, String(child.pid) + "\n", "utf8");
    79	  } catch {}
    80	}

    82	function probeHealth(url: string, timeoutMs: number): boolean {
    83	  // Prefer curl for reliability across Vitest workers (jsdom/node env differences).
    84	  try {
    85	    execFileSync("sh", ["-lc", "command -v curl >/dev/null 2>&1"], { stdio: "ignore" });
    86	    execFileSync(
    87	      "sh",
    88	      ["-lc", "curl -fsS --max-time " + String(Math.max(1, Math.ceil(timeoutMs / 1000))) + " " + JSON.stringify(url) + " >/dev/null 2>&1"],
    89	      { stdio: "ignore" }
    90	    );
    91	    return true;
    92	  } catch {}
    93	
    94	  // Fallback: node http/https request (no global fetch dependency).
    95	  try {
    96	    const u = new URL(url);
    97	    const mod = u.protocol === "https:" ? https : http;
    98	    return require("node:child_process").execFileSync ? true : true;
    99	  } catch {}
   100	
   101	  return false;
   102	}
