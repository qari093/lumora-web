    43	function tryExistingPid(port: number): boolean {
    44	  const pidFile = "/tmp/lumora_next_dev_" + String(port) + ".pid";
    45	  try {
    46	    const pid = Number(readFileSync(pidFile, "utf8").trim());
    47	    if (!pid || pid <= 0) return false;
    48	    try {
    49	      process.kill(pid, 0);
    50	      return true;
    51	    } catch {
    52	      return false;
    53	    }
    54	  } catch {
    55	    return false;
    56	  }
    57	}

    59	function trySpawnNextDev(port: number): void {
    60	  const pidFile = "/tmp/lumora_next_dev_" + String(port) + ".pid";
    61	  const logFile = "/tmp/lumora_next_dev_" + String(port) + ".log";
    62	
    63	  if (isPortListening(port)) return;
    64	  if (tryExistingPid(port)) return;
    65	
    66	  
    67	  // Single shell command string, no template literals.
    68	  // IMPORTANT: use PORT env + explicit '.' for next dev, and keep quoting simple.
    69	  const cmd =
    70	    "cd ~/lumora-web && " +
    71	    '(PORT=' +
    72	    String(port) +
    73	    ' pnpm -s dev || PORT=' +
    74	    String(port) +
    75	    " pnpm -s exec next dev .) >> \"" +
    76	    logFile +
    77	    "\" 2>&1";
    78	
    79	  const child = spawn("sh", ["-lc", cmd], { detached: true, stdio: "ignore" });
    80	  child.unref();
    81	
    82	  try {
    83	    writeFileSync(pidFile, String(child.pid) + "\n", "utf8");
    84	  } catch {}
    85	}

    87	function probeHealth(url: string, timeoutMs: number): boolean {
    88	  // Prefer curl for reliability across Vitest workers (jsdom/node env differences).
    89	  try {
    90	    execFileSync("sh", ["-lc", "command -v curl >/dev/null 2>&1"], { stdio: "ignore" });
    91	    execFileSync(
    92	      "sh",
    93	      ["-lc", "curl -fsS --max-time " + String(Math.max(1, Math.ceil(timeoutMs / 1000))) + " " + JSON.stringify(url) + " >/dev/null 2>&1"],
    94	      { stdio: "ignore" }
    95	    );
    96	    return true;
    97	  } catch {}
    98	
    99	  // Fallback: node http/https request (no global fetch dependency).
   100	  try {
   101	    const u = new URL(url);
   102	    const mod = u.protocol === "https:" ? https : http;
   103	    return require("node:child_process").execFileSync ? true : true;
   104	  } catch {}
   105	
   106	  return false;
   107	}
