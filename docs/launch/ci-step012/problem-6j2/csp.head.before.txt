     1	import { describe, expect, test } from "vitest";
     2	
     3	const BASE = new URL(process.env.LUMORA_TEST_BASE || "http://127.0.0.1:3000");
     4	
     5	// Portal sweeps can be slow on first compile; keep this scoped to security suite only.
     6	const PORTAL_HEAD_TIMEOUT_MS = 25_000;
     7	
     8	async function head(path: string, timeoutMs = PORTAL_HEAD_TIMEOUT_MS): Promise<{ status: number; headers: Headers }> {
     9	  const url = new URL(path, BASE);
    10	  const ac = new AbortController();
    11	  const t = setTimeout(() => ac.abort(new Error("aborted")), timeoutMs);
    12	  const started = Date.now();
    13	  try {
    14	    const res = await fetch(url, { method: "HEAD", cache: "no-store", signal: ac.signal });
    15	    return { status: res.status, headers: res.headers };
    16	  } catch (e: any) {
    17	    const ms = Date.now() - started;
    18	    const msg = typeof e?.message === "string" ? e.message : "fetch_failed";
    19	    throw new Error(`HEAD failed: ${path} (${ms}ms) — ${msg}`);
    20	  } finally {
    21	    clearTimeout(t);
    22	  }
    23	}
    24	
    25	function getHeader(h: Headers, name: string): string {
    26	  return (h.get(name) || "").toString();
    27	}
    28	
    29	describe("security headers: CSP smoke", () => {
    30	  test(
    31	    "sets Content-Security-Policy on root and core portals (pinpoint slow route)",
    32	    async () => {
    33	      const paths = ["/", "/gmar", "/movies", "/nexa", "/videos", "/video", "/live"];
    34	      for (const p of paths) {
    35	        const r = await head(p);
    36	        expect([200, 301, 302, 307, 308]).toContain(r.status);
    37	        const csp = getHeader(r.headers, "content-security-policy");
    38	        expect(csp.length).toBeGreaterThan(0);
    39	      }
    40	    },
    41	    240_000
    42	  );
    43	
    44	  test(
    45	    "sets Content-Security-Policy on /api/health",
    46	    async () => {
    47	      const r = await head("/api/health", 35_000);
    48	      expect([200, 301, 302, 307, 308]).toContain(r.status);
    49	      const csp = getHeader(r.headers, "content-security-policy");
    50	      expect(csp.length).toBeGreaterThan(0);
    51	    },
    52	    120_000
    53	  );
    54	});
