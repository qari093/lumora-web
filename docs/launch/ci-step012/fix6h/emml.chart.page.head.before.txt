     1	"use client";
     2	
     3	import * as React from "react";
     4	import { fetchEmmlChart } from "@/app/_client/emml-chart-client";
     5	
     6	type ChartState = any;
     7	
     8	export default function EmmlChartPage(): JSX.Element {
     9	  const [data, setData] = React.useState<ChartState>(null);
    10	  const [err, setErr] = React.useState<string | null>(null);
    11	
    12	  React.useEffect(() => {
    13	    let alive = true;
    14	    (async () => {
    15	      try {
    16	        const res = await fetchEmmlChart();
    17	        if (alive) setData(res);
    18	      } catch (e: any) {
    19	        if (alive) setErr(e?.message ?? "failed");
    20	      }
    21	    })();
    22	    return () => { alive = false; };
    23	  }, []);
    24	
    25	  if (err) {
    26	    return (
    27	      <div style={{ padding: 16, fontFamily: "system-ui" }}>
    28	        <h1>EMML Chart</h1>
    29	        <p style={{ opacity: 0.8 }}>Error loading chart: {err}</p>
    30	      </div>
    31	    );
    32	  }
    33	
    34	  if (!data) {
    35	    return (
    36	      <div style={{ padding: 16, fontFamily: "system-ui" }}>
    37	        <h1>EMML Chart</h1>
    38	        <p style={{ opacity: 0.8 }}>Loading…</p>
    39	      </div>
    40	    );
    41	  }
    42	
    43	  return (
    44	    <div style={{ padding: 16, fontFamily: "system-ui" }}>
    45	      <h1>EMML Chart</h1>
    46	      <pre style={{ whiteSpace: "pre-wrap", wordBreak: "break-word" }}>
    47	        {JSON.stringify(data, null, 2)}
    48	      </pre>
    49	    </div>
    50	  );
    51	}
