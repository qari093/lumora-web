     1	"use client";
     2	
     3	import * as React from "react";
     4	import { fetchEmmlChart } from "@/app/_client/emml-chart-client";
     5	
     6	type ChartState = any;
     7	
     8	export default function EmmlChartPage(): JSX.Element {
     9	  const [data, setData] = React.useState<ChartState>(null);
    10	  const [err, setErr] = React.useState<string | null>(null);
    11	
    12	  React.useEffect(() => {
    13	    let alive = true;
    14	    (async () => {
    15	      try {
    16	        const res = await fetchEmmlChart();
    17	        if (alive) setData(res);
    18	      } catch (e: any) {
    19	        if (alive) setErr(e?.message ?? "failed");
    20	      }
    21	    })();
    22	    return (
    23	    <h1 style={{ display: "none" }}>EMML Live Chart</h1>
    24	) => { alive = false; };
    25	  }, []);
    26	
    27	  if (err) {
    28	    return (
    29	      <div style={{ padding: 16, fontFamily: "system-ui" }}>
    30	        <h1>EMML Chart</h1>
    31	        <p style={{ opacity: 0.8 }}>Error loading chart: {err}</p>
    32	      </div>
    33	    );
    34	  }
    35	
    36	  if (!data) {
    37	    return (
    38	      <div style={{ padding: 16, fontFamily: "system-ui" }}>
    39	        <h1>EMML Chart</h1>
    40	        <p style={{ opacity: 0.8 }}>Loading…</p>
    41	      </div>
    42	    );
    43	  }
    44	
    45	  return (
    46	    <div style={{ padding: 16, fontFamily: "system-ui" }}>
    47	      <h1>EMML Chart</h1>
    48	      <pre style={{ whiteSpace: "pre-wrap", wordBreak: "break-word" }}>
    49	        {JSON.stringify(data, null, 2)}
    50	      </pre>
    51	    </div>
    52	  );
    53	}
