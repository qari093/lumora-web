import { describe, it, expect, beforeAll } from "vitest";

const BASE =
  (process.env.BASE_URL && process.env.BASE_URL.trim()) ||
  "http://127.0.0.1:3000";

const HOOK_TIMEOUT_MS = 120_000;
const TEST_TIMEOUT_MS = 60_000;

async function sleep(ms: number) {
  return new Promise((r) => setTimeout(r, ms));
}

async function fetchText(path: string, timeoutMs = 8_000) {
  const ac = new AbortController();
  const t = setTimeout(() => ac.abort(new Error(`abort:${timeoutMs}ms`)), timeoutMs);
  try {
    const res = await fetch(new URL(path, BASE), {
      cache: "no-store",
      redirect: "manual",
      signal: ac.signal,
    });
    const text = await res.text();
    return { res, text };
  } finally {
    clearTimeout(t);
  }
}

async function ensureHealth(maxWaitMs = 75_000) {
  const start = Date.now();
  let lastErr = "unknown";
  while (Date.now() - start < maxWaitMs) {
    try {
      const { res, text } = await fetchText("/api/health", 5_000);
      // If Next is down you'll often get ECONNREFUSED (caught) or an HTML page.
      if (res.ok) {
        // Accept JSON only. HTML indicates route not mounted or rewrites interfering.
        const ct = res.headers.get("content-type") ?? "";
        if (ct.includes("application/json")) {
          try {
            const j = JSON.parse(text);
            if (j && j.ok === true) return;
            lastErr = `health_json_not_ok:${text.slice(0, 120)}`;
          } catch {
            lastErr = `health_json_parse_fail:${text.slice(0, 120)}`;
          }
        } else {
          lastErr = `health_not_json:${ct}:${text.slice(0, 120)}`;
        }
      } else {
        lastErr = `health_status_${res.status}:${text.slice(0, 120)}`;
      }
    } catch (e: any) {
      lastErr = typeof e?.message === "string" ? e.message : "fetch_failed";
    }
    await sleep(350);
  }
  throw new Error(`Test server not healthy within timeout. last=${lastErr}`);
}

describe("health smoke", () => {
  beforeAll(async () => {
    await ensureHealth();
  }, HOOK_TIMEOUT_MS);

  it(
    "/api/health responds and is json (smoke, robust)",
    async () => {
      const { res, text } = await fetchText("/api/health", 8_000);
      expect(res.ok).toBe(true);
      const ct = res.headers.get("content-type") ?? "";
      expect(ct).toContain("application/json");
      const j = JSON.parse(text);
      expect(j.ok).toBe(true);
    },
    TEST_TIMEOUT_MS
  );
});
