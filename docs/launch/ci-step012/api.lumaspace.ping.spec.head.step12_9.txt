import { describe, it, expect } from "vitest";
import { fetchTextRobust } from "./helpers/http";

describe("LumaSpace API — ping (integration)", () => {
  it(
    "GET /api/lumaspace/ping returns ok JSON",
    async () => {
      const res = await fetchTextRobust("/api/lumaspace/ping", { timeoutMs: 8000, tries: 8, backoffMs: 160 });
      expect(res.ok).toBe(true);
      expect(res.ct.toLowerCase()).toContain("application/json");
      const obj = JSON.parse(res.text);
      // contract: { ok: true, ... } or { status: "ok", ... }
      const ok = obj?.ok === true || obj?.status === "ok" || obj?.alive === true;
      expect(ok).toBe(true);
    },
    { timeout: 20000 }
  );
});
