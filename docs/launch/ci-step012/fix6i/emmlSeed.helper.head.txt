     1	import { PrismaClient } from "@prisma/client";
     2	
     3	/**
     4	 * Best-effort EMML seeding for tests.
     5	 * Strategy:
     6	 *  1) If DB already has EMML data, return { seeded: true }.
     7	 *  2) Attempt to call an existing app seed utility (if present) via dynamic import.
     8	 *  3) Re-check DB; if still empty, return { seeded: false } so tests can skip deterministically.
     9	 */
    10	export async function ensureEmmlSeed(): Promise<{ seeded: boolean; reason?: string }> {
    11	  const db = new PrismaClient();
    12	
    13	  async function counts() {
    14	    const [idx, mkt, ast, tick] = await Promise.all([
    15	      (db as any).emmlIndex?.count?.().catch(() => 0) ?? 0,
    16	      (db as any).emmlMarket?.count?.().catch(() => 0) ?? 0,
    17	      (db as any).emmlAsset?.count?.().catch(() => 0) ?? 0,
    18	      (db as any).emmlTick?.count?.().catch(() => 0) ?? 0,
    19	    ]);
    20	    return { idx, mkt, ast, tick };
    21	  }
    22	
    23	  try {
    24	    const c0 = await counts();
    25	    if (c0.idx > 0 && c0.mkt > 0 && c0.ast > 0 && c0.tick > 0) {
    26	      return { seeded: true };
    27	    }
    28	
    29	    // Try a set of likely seed utility modules / functions.
    30	    const candidates: Array<{ mod: string; fns: string[] }> = [
    31	      { mod: "../app/emml/seed", fns: ["seedEmml", "seed", "ensureSeed", "seedDemo", "seedEmmlDemo"] },
    32	      { mod: "../app/_lib/emml/seed", fns: ["seedEmml", "seed", "ensureSeed", "seedDemo", "seedEmmlDemo"] },
    33	      { mod: "../app/_lib/emml/seedDemo", fns: ["seedEmmlDemo", "seedDemo", "seed"] },
    34	      { mod: "../app/_lib/emml/demoSeed", fns: ["seedEmmlDemo", "seedDemo", "seed"] },
    35	      { mod: "../app/_lib/emml", fns: ["seedEmml", "seedEmmlDemo", "seedDemo"] },
    36	      { mod: "../app/_lib/emml/index", fns: ["seedEmml", "seedEmmlDemo", "seedDemo"] },
    37	      { mod: "../app/api/emml/seed/route", fns: ["POST", "GET"] },
    38	    ];
    39	
    40	    let invoked = false;
    41	    for (const c of candidates) {
    42	      try {
    43	        const m: any = await import(c.mod);
    44	        for (const fnName of c.fns) {
    45	          const fn = m?.[fnName];
    46	          if (typeof fn === "function") {
    47	            // If it's a Next route handler, call it without req (handlers should tolerate).
    48	            if (fnName === "GET" || fnName === "POST") {
    49	              await fn(new Request("http://127.0.0.1/api/emml/seed", { method: fnName }));
    50	            } else {
    51	              await fn();
    52	            }
    53	            invoked = true;
    54	            break;
    55	          }
    56	        }
    57	        if (invoked) break;
    58	      } catch {
    59	        // ignore
    60	      }
    61	    }
    62	
    63	    const c1 = await counts();
    64	    if (c1.idx > 0 && c1.mkt > 0 && c1.ast > 0 && c1.tick > 0) {
    65	      return { seeded: true };
    66	    }
    67	
    68	    return {
    69	      seeded: false,
    70	      reason: invoked
    71	        ? "seed_util_ran_but_db_still_empty"
    72	        : "no_seed_util_found_and_db_empty",
    73	    };
    74	  } finally {
    75	    await db.$disconnect().catch(() => {});
    76	  }
    77	}
