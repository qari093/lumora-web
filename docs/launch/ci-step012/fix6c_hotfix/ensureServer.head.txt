import { execFileSync, spawn } from "node:child_process";
import { readFileSync, writeFileSync } from "node:fs";

type EnsureOpts = {
  baseUrl?: string;
  maxWaitMs?: number;
  pollMs?: number;
  port?: number;
};

function sanitizeBaseUrl(raw: string | undefined, port: number): string {
  const v = (raw || "").trim();
  if (!v) return \`http://127.0.0.1:\${port}\`;
  if (!/^https?:\\/\\//i.test(v)) return \`http://\${v}\`;
  return v;
}

function isPortListening(port: number): boolean {
  try {
    execFileSync("sh", ["-lc", \`lsof -nP -iTCP:\${port} -sTCP:LISTEN >/dev/null 2>&1\`], { stdio: "ignore" });
    return true;
  } catch {
    return false;
  }
}

function tryExistingPid(port: number): boolean {
  const pidFile = \`/tmp/lumora_next_dev_\${port}.pid\`;
  try {
    const pid = Number(readFileSync(pidFile, "utf8").trim());
    if (!pid || pid <= 0) return false;
    try {
      process.kill(pid, 0);
      return true;
    } catch {
      return false;
    }
  } catch {
    return false;
  }
}

function trySpawnNextDev(port: number): void {
  const pidFile = \`/tmp/lumora_next_dev_\${port}.pid\`;
  const logFile = \`/tmp/lumora_next_dev_\${port}.log\`;

  if (isPortListening(port)) return;
  if (tryExistingPid(port)) return;

  // Build a single shell command string (no nested template literals in generator).
  const shell = [
    "cd ~/lumora-web",
    // Prefer pnpm dev; fallback to pnpm exec next dev.
    \`(pnpm -s dev -- -p \${port} || pnpm -s exec next dev -p \${port}) >> "\${logFile}" 2>&1\`,
  ].join(" && ");

  const child = spawn("sh", ["-lc", shell], { detached: true, stdio: "ignore" });
  child.unref();

  try {
    writeFileSync(pidFile, String(child.pid) + "\\n", "utf8");
  } catch {}
}

async function fetchOk(url: string, timeoutMs: number): Promise<boolean> {
  const ac = new AbortController();
  const t = setTimeout(() => ac.abort(), timeoutMs);
  try {
    const r = await fetch(url, { cache: "no-store", signal: ac.signal });
    if (!r.ok) return false;
    const ct = (r.headers.get("content-type") || "").toLowerCase();
    if (ct.includes("application/json")) {
      const j = await r.json().catch(() => null);
      return !!j;
    }
    const txt = await r.text().catch(() => "");
    return txt.length > 0;
  } catch {
    return false;
  } finally {
    clearTimeout(t);
  }
}

export async function ensureServerReady(opts: EnsureOpts = {}): Promise<string> {
  const port = Number(opts.port || process.env.PORT || 3000);
  const base = sanitizeBaseUrl(opts.baseUrl || process.env.BASE_URL, port);

  // Publish sanitized BASE_URL for all tests.
  process.env.BASE_URL = base;

  const maxWaitMs = Number(opts.maxWaitMs ?? 20_000);
  const pollMs = Number(opts.pollMs ?? 250);

  const health = new URL("/api/health", base).toString();

  if (!isPortListening(port)) {
    trySpawnNextDev(port);
  }

  const start = Date.now();
  while (Date.now() - start < maxWaitMs) {
    if (!isPortListening(port)) {
      trySpawnNextDev(port);
    }
    if (await fetchOk(health, 1200)) return base;
    await new Promise((r) => setTimeout(r, pollMs));
  }

  throw new Error(\`Test server not healthy within timeout (\${maxWaitMs}ms) @ \${base}\`);
}
